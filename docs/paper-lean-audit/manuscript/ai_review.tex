% Shared review markup and formal links; the human draft hides declaration names.
% Review annotations, following the red-add/blue-remove convention used in
% the Varenna 2016 papers. These commands keep the original wording visible.
% Accept an insertion by removing its AIaddition environment or AIadd wrapper.
% Accept a replacement by deleting AIremove{old} and unwrapping AIadd{new}.
% Resolve a comment by deleting its complete AIcomment{ID}{text} command.
% There is deliberately no global switch that hides unresolved comments.
\usepackage{xcolor}
\usepackage{xurl}
\usepackage[normalem]{ulem}
% AI compile repair: the retained draft uses the undefined typo \cots.
% EDIT-02 beside that occurrence makes this repair visible in the PDF.
\providecommand{\cots}{\cdots}
\definecolor{AIadded}{RGB}{160,25,25}
\definecolor{AIreview}{RGB}{25,70,145}
\newcommand{\AIadd}[1]{{\color{AIadded}#1}}
\newcommand{\AIremove}[1]{{\color{AIreview}\sout{#1}}}
\newcommand{\AIreplace}[2]{\AIremove{#1}\AIadd{#2}}
\newenvironment{AIaddition}[1]
  {\par\begingroup\color{AIadded}\noindent
   \textbf{AI addition: #1.}\par\smallskip\ignorespaces}
  {\par\endgroup}
\newcommand{\AIcomment}[2]{%
  \par\smallskip\noindent
  \fcolorbox{AIreview}{AIreview!4}{%
    \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
      \normalfont\small\color{AIreview}\textbf{AI comment #1.} #2}}%
  \par\smallskip}
\newcommand{\AIreviewlegend}{%
  \AIcomment{Review guide}{\textbf{Suggested changes.} Red text is proposed
  wording, not an accepted correction. Blue strike-through preserves the
  original text. Accept or revise the proposal before removing its wrappers.
  \textbf{Comments and action.} Every box states its status: \emph{Proposed
  change} awaits author acceptance; \emph{Action required} needs a proof repair
  or drafting decision; \emph{Information} requests no manuscript edit;
  \emph{Optional refinement} may be omitted. A different verified proof does
  not close a gap in the printed proof.
  \textbf{Who reviews.} Johanna can edit wording and integrate agreed changes;
  Max reviews the whole paper--Lean correspondence, including exact matches,
  definitions and printed proof steps. He checks mathematical gaps before
  asking Johanna for a text repair. Authors decide
  the geometric proof route. Follow \emph{Mathematical proof} links for the
  readable comparison; no Lean code is needed. Resolve a box only after its
  stated task is done, by deleting its \texttt{\string\AIcomment} command.}}
